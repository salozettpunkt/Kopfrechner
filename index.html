<button onclick="addPlayer()">Spieler hinzufügen</button>

<button onclick="addRound()">Neue Runde</button>


<script>

  let players = [];

  let roundCount = 0;


  function addPlayer(name = `Spieler ${players.length + 1}`) {

    players.push({ name });


    // Header

    const th = document.createElement("th");

    th.innerHTML = `<input value="${name}" oninput="renamePlayer(${players.length - 1}, this.value)">`;

    document.getElementById("headerRow").appendChild(th);


    // Total

    const td = document.createElement("td");

    td.id = `total-${players.length - 1}`;

    td.innerText = "0";

    document.getElementById("totalRow").appendChild(td);


    // Existing rounds

    document.querySelectorAll("#rounds tr").forEach(row => {

      const cell = document.createElement("td");

      cell.innerHTML = `<input type="number" value="0" oninput="updateTotals()">`;

      row.appendChild(cell);

    });

  }


  function renamePlayer(index, newName) {

    players[index].name = newName;

  }


  function addRound() {

    roundCount++;

    const tr = document.createElement("tr");

    tr.innerHTML = `<td>${roundCount}</td>`;


    players.forEach(() => {

      const td = document.createElement("td");

      td.innerHTML = `<input type="number" value="0" oninput="updateTotals()">`;

      tr.appendChild(td);

    });


    document.getElementById("rounds").appendChild(tr);

  }


  function updateTotals() {

    const target = Number(document.getElementById("targetScore").value);


    players.forEach((_, index) => {

      let sum = 0;

      document.querySelectorAll(`#rounds tr td:nth-child(${index + 2}) input`)

        .forEach(input => sum += Number(input.value) || 0);


      const totalCell = document.getElementById(`total-${index}`);

      totalCell.innerText = sum;


      if (target && sum >= target) {

        totalCell.classList.add("winner");

      } else {

        totalCell.classList.remove("winner");

      }

    });

  }


  // Startzustand

  addPlayer();

  addPlayer();

  addRound();

</script>
